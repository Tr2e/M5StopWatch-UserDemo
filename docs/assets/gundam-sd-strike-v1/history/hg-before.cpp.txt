constexpr float pi=3.14159265359f;
constexpr uint16_t white=0xf7be,ivory=0xdedb,blue=0x2299,frame=0x5aeb,black=0x18c4;
constexpr uint16_t gold=0xfe67,red=0xe966,cyan=0x3dff;
struct Ring {float y,w,d,z=0;};
class StrikeBuilder {
public:
    Mesh& m;BuildOptions options;
    Part part=Part::Torso;Point origin{};float roll=0,pitch=0,yaw=0;
    void at(Part p,Point o={},float r=0,float t=0,float a=0){part=p;origin=o;roll=r;pitch=t;yaw=a;}
    Point transform(Point p)const{
        const float cp=std::cos(pitch),sp=std::sin(pitch),cy=std::cos(yaw),sy=std::sin(yaw);
        p={p.x,p.y*cp-p.z*sp,p.y*sp+p.z*cp};
        p={p.x*cy+p.z*sy,p.y,p.z*cy-p.x*sy};
        const float c=std::cos(roll),s=std::sin(roll);
        p={origin.x+p.x*c-p.y*s,origin.y+p.x*s+p.y*c,origin.z+p.z};
        if(part==Part::Torso){p.y=2.82f+(p.y-2.82f)*.86f;p.x*=1.07f;}
        if(part==Part::Waist)p.y+=.10f;
        if(part==Part::Thighs)p.y=1.34f+(p.y-1.34f)*1.16f;
        return p;
    }
    void face(Point a,Point b,Point c,Point d,uint16_t color,Point outward,bool insert=false){
        auto n=cross(subtract(b,a),subtract(c,a));
        const float len=std::sqrt(dot(n,n));if(len<1e-8f)return;
        if(std::abs(dot(n,subtract(d,a)))>1e-6f*len){
            face(a,b,c,c,color,outward,insert);face(a,c,d,d,color,outward,insert);return;
        }
        if(dot(n,outward)<0){
            if(c.x==d.x&&c.y==d.y&&c.z==d.z){std::swap(b,c);d=c;}else std::swap(b,d);
        }
        a=transform(a);b=transform(b);c=transform(c);d=transform(d);
        n=cross(subtract(b,a),subtract(c,a));const float length=std::sqrt(dot(n,n));
        if(length<1e-8f)return;
        if(m.count>=Mesh::capacity){m.overflowed=true;return;}
        n={n.x/length,n.y/length,n.z/length};
        const float light=.56f+.36f*std::max(0.f,dot(n,Point{-.46f,.65f,.60f}))
            +.13f*std::max(0.f,dot(n,Point{.70f,.25f,-.67f}));
        lets_and_go::mesh_parts::MeshWriter writer{{m.panels.data(),Mesh::capacity},m.count,false};
        lets_and_go::mesh_parts::Builder base{writer,12};
        base.quad(a,b,c,d,lets_and_go::mesh_parts::shade(options.gray?ivory:color,light));
        m.normals[m.count]=n;m.parts[m.count]=part;m.twoSided[m.count]=insert || part==Part::Head;m.count=writer.count;
    }
    void box(float x,float y,float z,float w,float h,float d,uint16_t color,bool openFront=false){
        const float l=x-w/2,r=x+w/2,b=y-h/2,t=y+h/2,f=z+d/2,k=z-d/2;
        if(!openFront)face({l,b,f},{r,b,f},{r,t,f},{l,t,f},color,{0,0,1});
        face({r,b,k},{l,b,k},{l,t,k},{r,t,k},color,{0,0,-1});
        face({l,b,k},{l,b,f},{l,t,f},{l,t,k},color,{-1,0,0});
        face({r,b,f},{r,b,k},{r,t,k},{r,t,f},color,{1,0,0});
        face({l,t,f},{r,t,f},{r,t,k},{l,t,k},color,{0,1,0});
        face({l,b,k},{r,b,k},{r,b,f},{l,b,f},color,{0,-1,0});
    }
    void shell(std::initializer_list<Ring> rings,uint16_t color,bool aperture=false,int count=8){
        const auto point=[&](Ring r,int i){
            if(count==8){
                constexpr float xs[]={-.72f,.72f,1,1,.72f,-.72f,-1,-1};
                constexpr float zs[]={1,1,.70f,-.70f,-1,-1,-.70f,.70f};
                return Point{xs[i%8]*r.w,r.y,r.z+zs[i%8]*r.d};
            }
            const float a=2*pi*i/count;
            return Point{std::sin(a)*r.w,r.y,r.z+std::cos(a)*r.d};
        };
        for(auto it=rings.begin()+1;it!=rings.end();++it){
            const auto a=*(it-1),b=*it;
            for(int i=0;i<count;++i){
                if(part==Part::Shins && count==16 && a.y>=-1.53f && b.y<=-1.12f && (i==7 || i==8)){
                    // The rear window is cut out of the shell, with real edge thickness.
                    const int edge=i==7?7:9;auto low=point(a,edge),high=point(b,edge);
                    const auto inside=[](Point p){return Point{p.x*.90f,p.y,p.z+.018f};};
                    face(low,high,inside(high),inside(low),ivory,{i==7?-1.f:1.f,0,0},true);
                    if(a.y==-1.53f)face(point(a,i),point(a,i+1),inside(point(a,i+1)),inside(point(a,i)),ivory,{0,1,0},true);
                    if(b.y==-1.12f)face(point(b,i),point(b,i+1),inside(point(b,i+1)),inside(point(b,i)),ivory,{0,-1,0},true);
                    continue;
                }
                if(aperture && a.y<.25f && (count==8?(i==0||i==1||i==7):(i<4||i>=count-4)))continue;
                auto p=point(a,i),q=point(a,i+1);
                face(p,q,point(b,i+1),point(b,i),color,{(p.x+q.x)/2,0,(p.z+q.z)/2-a.z},aperture);
            }
        }
        for(int end=0;end<2;++end){const auto r=end?*(rings.end()-1):*rings.begin();
            for(int i=1;i<count-1;i+=2)face(point(r,0),point(r,i),point(r,i+1),point(r,i+2),color,{0,end?1.f:-1.f,0},aperture);
        }
    }
    void opening(Point a,Point b,Point c,Point d,float depth,uint16_t border){
        const Point center{(a.x+b.x+c.x+d.x)/4,(a.y+b.y+c.y+d.y)/4,(a.z+b.z+c.z+d.z)/4};
        const Point outer[]={a,b,c,d};Point inner[4],back[4];
        for(int i=0;i<4;++i){const auto p=outer[i];
            inner[i]={center.x+(p.x-center.x)*.82f,center.y+(p.y-center.y)*(part==Part::Torso?.56f:.72f),center.z+(p.z-center.z)*.72f};
            if(part==Part::Torso){
                const float rise=(std::abs(inner[i].x)-.16f)*.18f;
                inner[i].y+=rise;inner[i].z-=rise*.26875f;
            }
            // Interpolate depth with vertical inset; this opening is on an inclined front plane.
            back[i]={inner[i].x,inner[i].y,inner[i].z-depth};
        }
        for(int i=0;i<4;++i){const int j=(i+1)%4;
            face(outer[i],outer[j],inner[j],inner[i],border,{0,0,1},true);
            face(inner[i],inner[j],back[j],back[i],border,{center.x-(inner[i].x+inner[j].x)/2,center.y-(inner[i].y+inner[j].y)/2,0},true);
        }
        face(back[0],back[1],back[2],back[3],black,{0,0,1},true);
        const auto lerp=[](Point x,Point y,float t){return Point{x.x*(1-t)+y.x*t,x.y*(1-t)+y.y*t,x.z*(1-t)+y.z*t+.005f};};
        if(part==Part::Torso){
            const float t=.52f;
            face(lerp(back[0],back[3],t),lerp(back[1],back[2],t),
                lerp(back[1],back[2],t+.08f),lerp(back[0],back[3],t+.08f),frame,{0,0,1},true);
        }
    }
    void plate(std::initializer_list<Point> points,float depth,uint16_t color,bool thin=false){
        Point center{};for(auto p:points){center.x+=p.x;center.y+=p.y;center.z+=p.z;}
        const float count=float(points.size());center={center.x/count,center.y/count,center.z/count};
        for(size_t i=0;i<points.size();++i){auto a=points.begin()[i],b=points.begin()[(i+1)%points.size()];
            face(center,a,b,b,color,{0,0,1},thin);
            face({center.x,center.y,center.z-depth},{a.x,a.y,a.z-depth},{b.x,b.y,b.z-depth},
                {b.x,b.y,b.z-depth},color,{0,0,-1},thin);
            face(a,{a.x,a.y,a.z-depth},{b.x,b.y,b.z-depth},b,color,{(a.x+b.x)/2-center.x,(a.y+b.y)/2-center.y,0},thin);
        }
    }
    // Bevelled cover: planar centre, independent border and sides. The outline must be convex and planar.
    void cover(std::initializer_list<Point> points,float depth,uint16_t color){
        Point mid{};for(auto q:points){mid.x+=q.x;mid.y+=q.y;mid.z+=q.z;}
        const float n=float(points.size());mid={mid.x/n,mid.y/n,mid.z/n};
        const auto inset=[&](Point p){return Point{mid.x+(p.x-mid.x)*.86f,mid.y+(p.y-mid.y)*.86f,mid.z+(p.z-mid.z)*.86f};};
        for(size_t i=0;i<points.size();++i){auto a=points.begin()[i],c=points.begin()[(i+1)%points.size()];
            auto ai=inset(a),ci=inset(c);a.z-=.014f;c.z-=.014f;
            face(mid,ai,ci,ci,color,{0,0,1});face(a,c,ci,ai,color,{0,0,1});
            auto aa=a,cc=c;aa.z-=depth;cc.z-=depth;
            face(a,aa,cc,c,color,{a.x+c.x-2*mid.x,a.y+c.y-2*mid.y,0});
            face({mid.x,mid.y,mid.z-depth-.014f},aa,cc,cc,ivory,{0,0,-1});
        }
    }
    void tube(Point a,Point b,float r0,float r1,uint16_t color,bool hollow=false,int count=10,bool buriedStart=false){
        auto axis=subtract(b,a);const float len=std::sqrt(dot(axis,axis));if(len<1e-6f)return;
        axis={axis.x/len,axis.y/len,axis.z/len};auto u=cross(axis,std::abs(axis.y)<.9f?Point{0,1,0}:Point{1,0,0});
        const float ul=std::sqrt(dot(u,u));u={u.x/ul,u.y/ul,u.z/ul};const auto v=cross(axis,u);
        const auto p=[&](Point c,float radius,int i){const float t=2*pi*i/count,co=std::cos(t),si=std::sin(t);
            return Point{c.x+radius*(u.x*co+v.x*si),c.y+radius*(u.y*co+v.y*si),c.z+radius*(u.z*co+v.z*si)};};
        for(int i=0;i<count;++i){auto x=p(a,r0,i),y=p(a,r0,i+1);
            face(x,y,p(b,r1,i+1),p(b,r1,i),color,subtract(x,a));
            if(!buriedStart || options.keepBuriedFaces)face(a,y,x,x,color,{-axis.x,-axis.y,-axis.z});
            else ++m.buriedOmitted;
            if(!hollow){face(b,p(b,r1,i),p(b,r1,i+1),p(b,r1,i+1),color,axis);continue;}
            const float recess=std::min(.045f,len*.65f);
            const auto inner=Point{b.x-axis.x*recess,b.y-axis.y*recess,b.z-axis.z*recess};
            face(p(b,r1,i),p(b,r1,i+1),p(b,r1*.70f,i+1),p(b,r1*.70f,i),color,axis);
            face(p(b,r1*.70f,i),p(b,r1*.70f,i+1),p(inner,r1*.66f,i+1),p(inner,r1*.66f,i),frame,subtract(b,p(b,r1,i)));
            face(inner,p(inner,r1*.66f,i),p(inner,r1*.66f,i+1),p(inner,r1*.66f,i+1),black,axis);
        }
    }
};

    for(float s:{-1.f,1.f}){

                p(.071f,-1.53f,.105f)},.009f,frame);

            for(int i=0;i<2;++i){
                const auto edge=[&](int j){return Point{side*ww[j]*.38268343f,yy[j],zz[j]+dd[j]*.92387953f};};

        b.opening({-.039f,-1.53f,.1243f},{.039f,-1.53f,.1243f},{.0532f,-1.12f,.1824f},{-.0532f,-1.12f,.1824f},.026f,frame);
        b.at(Part::Shins,{s*.235f,1.91f,0},s*.14f);
        // Open U ankle cuff (A23/A24); the top and inner walls are visible.

            b.tube({side*.10f,-1.605f,-.016f},{side*.161f,-1.605f,-.016f},.044f,.044f,ivory,false,8);
        }

            {0,-.89f,.226f},{-.107f,-.75f,.209f}},.051f,white);

        // The raised top slot is an actual open bridge over the carrier.
        b.box(-.115f,.218f,0,.034f,.045f,.236f,frame);

            {.085f,-.735f,.142f},{-.085f,-.735f,.142f},{-.098f,-.533f,.1289f}},.020f,white);

    b.at(Part::Waist);b.shell({{1.84f,.24f,.11f},{2.045f,.28f,.151f}},black);

    b.plate({{-.042f,2.06f,.19f},{.042f,2.06f,.19f},{.035f,1.973f,.20f},{-.035f,1.973f,.20f}},.012f,blue);
    b.plate({{-.043f,1.885f,.211f},{.043f,1.885f,.211f},{.034f,1.73f,.229f},{-.034f,1.73f,.229f}},.018f,frame);
    b.at(Part::Torso);
    b.shell({{2.09f,.219f,.133f},{2.27f,.24f,.164f},{2.42f,.28f,.177f},{2.51f,.30f,.181f}},red);
    for(float s:{-1.f,1.f}){

    b.at(Part::Torso);

    const Ring rings[]={{2.47f,.286f,.18f},{2.57f,.362f,.208f},{2.73f,.35f,.165f},{2.82f,.268f,.118f}};

    const auto p=[&](Ring r,int i){return Point{xs[i%8]*r.w,r.y,zs[i%8]*r.d};};
    for(int row=0;row<3;++row)for(int i=0;i<8;++i){

    for(int end=0;end<2;++end)for(int i=1;i<7;i+=2)
        b.face(p(rings[end?3:0],0),p(rings[end?3:0],i),p(rings[end?3:0],i+1),p(rings[end?3:0],i+2),blue,{0,end?1.f:-1.f,0});

void head(StrikeBuilder& b){
    b.at(Part::Head,{0,3.005f,0});
    // A30/A31 shell is open in front; B15 eye/chin core and A4 mask fill that opening.
    const Ring levels[]={{-.123f,.145f,.098f,-.034f},{-.064f,.194f,.137f,-.034f},
        {.065f,.211f,.162f,-.034f},{.173f,.191f,.158f,-.033f},{.251f,.13f,.102f,-.026f},{.276f,.054f,.052f,-.024f}};
    constexpr int n=24;
    const auto p=[&](Ring r,int i){float a=2*pi*i/n;return Point{std::sin(a)*r.w,r.y,r.z+std::cos(a)*r.d};};
    for(int row=0;row<5;++row)for(int i=0;i<n;++i){
        if(row<2 && (i<4||i>=20))continue;
        auto a=p(levels[row],i),c=p(levels[row],i+1);
        b.face(a,c,p(levels[row+1],i+1),p(levels[row+1],i),white,{a.x+c.x,0,a.z+c.z+.068f},row<2);
    }
    for(int i=0;i<n;++i){auto a=p(levels[5],i),c=p(levels[5],i+1);
        b.face({0,.283f,-.024f},a,c,c,white,{0,1,0});}
    b.tube({0,-.15f,-.023f},{0,-.095f,-.023f},.068f,.07f,black,false,10);
    // A continuous lower rim and inner wall support the open face shell.
    for(int i=4;i<20;++i){
        auto a=p(levels[0],i),c=p(levels[0],i+1);
        const auto q=[](Point v){return Point{v.x*.88f,v.y+.024f,-.034f+(v.z+.034f)*.86f};};
        b.face(a,c,q(c),q(a),ivory,{0,-1,0},true);
        auto aa=p(levels[2],i),cc=p(levels[2],i+1);
        b.face(q(a),q(c),q(cc),q(aa),ivory,{-a.x,0,-a.z-.034f},true);
    }
    // A30 cheek front is a narrow planar pillar, with a separate folded side cover.
    for(float s:{-1.f,1.f}){
        b.at(Part::Head,{0,3.005f,0});
        b.plate({{s*.088f,.026f,.176f},{s*.125f,.039f,.176f},{s*.129f,-.079f,.176f},

        b.face({s*.173f,.039f,.119f},{s*.173f,.039f,.01f},{s*.194f,.065f,-.009f},
            {s*.171f,.065f,.113f},white,{0,1,0},true);
        b.at(Part::Head,{s*.165f,3.06f,.14f});
        b.box(0,0,-.001f,.045f,.049f,.030f,white);
